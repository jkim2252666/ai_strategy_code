# 간단한 의사결정 나무를 통해 효과가 높은 집단의 특징 추출
library(rpart)
library(rpart.plot)
## Warning: package 'rpart.plot' was built under R version 4.5.2
a <- read_csv("marketing_campaign.csv")
## Rows: 2240 Columns: 30
## ── Column specification ────────────────────────────────────────────────────────
## Delimiter: ","
## chr   (2): Education, Marital_Status
## dbl  (27): Unnamed: 0, ID, Year_Birth, Income, Kidhome, Teenhome, Recency, M...
## date  (1): Dt_Customer
## 
## ℹ Use `spec()` to retrieve the full column specification for this data.
## ℹ Specify the column types or set `show_col_types = FALSE` to quiet this message.
a <- a %>%
  mutate(
    TotalSpent = MntWines + MntFruits + MntMeatProducts + 
      MntFishProducts + MntSweetProducts + MntGoldProds,
    child = Kidhome+Teenhome,
    age = 2025 - Year_Birth)

# outlier
a<-a[-2234,]

# 데이터 준비
# 처리(Treatment): Response (0 또는 1)
# 결과(Outcome): log(TotalSpent)
# 조건 변수(Covariates): Income, Age, Child 등 효과를 조절할 후보들
X <- as.matrix(a[, c("Income", "age", "child")]) 
Y <- log(a$TotalSpent)
W <- a$Response

# Causal Forest 모델 학습
# 이 모델은 Income, Age, Child의 복합적인 상호작용을 자동으로 찾아냅니다.
cf <- causal_forest(X, Y, W, num.trees = 2000)

# 개별 고객별 CATE(개인화된 효과) 추정
# 모든 데이터 포인트에 대해 '이 사람에게 캠페인이 미칠 영향'을 계산합니다.
cate_estimates <- predict(cf, X)$predictions
a$cate <- cate_estimates

# 결과 확인: 변수 중요도 (어떤 변수가 효과 차이를 가장 많이 만드는가?)
var_imp <- variable_importance(cf)
rownames(var_imp) <- colnames(X)
print(var_imp)
##             [,1]
## Income 0.5754570
## age    0.1828201
## child  0.2417229
# CATE를 종속변수로 하여 어떤 조건에서 효과가 극대화되는지 시각화
tree_model <- rpart(cate ~ Income + age + child, data = a)
rpart.plot(tree_model, main = "CATE 기반 타겟 고객 세분화")

# 자녀 수(child)에 따른 흑백/그레이스케일 그래프 (그림 3.43 참조)
ggplot(a, aes(x = Income, y = cate, 
              color = as.factor(child), 
              linetype = as.factor(child),
              shape = as.factor(child))) +
  # 개별 점 (흑백 인쇄 시 시각적 혼란을 줄이기 위해 투명도 낮춤)
  geom_point(alpha = 0.2, size = 1.5) +
  # 추세선 (se = FALSE로 신뢰구간 띠를 제거하면 선 구분이 훨씬 명확)
  geom_smooth(method = "loess", se = FALSE, size = 1) +
  # 그레이스케일 색상 팔레트 적용 (검은색 ~ 연한 회색)
  scale_color_grey(start = 0.1, end = 0.65) +
  theme_bw() +
  labs(
    title = "소득과 자녀수 조건별 CATE",
    x = "소득",
    y = "조건부 평균 처치 효과(CATE)",
    color = "자녀수",
    linetype = "자녀수",
    shape = "자녀수"
  )

# 나이를 해석하기 쉽게 3개 구간으로 범주화
a <- a %>%
  mutate(AgeGroup = cut(age, 
                        breaks = c(0, 35, 55, Inf), 
                        labels = c("청년 (35세 이하)", "중년 (35-55)", "시니어 (55세 이상)")))

# 다차원 시각화 실행 (출판용 흑백/그레이스케일 버전)
ggplot(a, aes(x = Income, y = cate, 
              # 색상 대신 회색 명암(color)과 선 모양(linetype)을 그룹에 매핑
              color = as.factor(child), 
              linetype = as.factor(child))) +
  # 개별 고객 데이터 (점): 흑백 인쇄 시 혼란을 줄이기 위해 투명도를 높임
  geom_point(alpha = 0.08, size = 1) + 
  # 그룹별 비선형 추세선: se = FALSE로 신뢰구간을 제거하여 선 구분을 명확히 함
  geom_smooth(method = "loess", se = FALSE, size = 1.2) + 
  # 나이대별로 그래프 분할
  facet_wrap(~ AgeGroup) +
  # 출판용 깔끔한 배경
  theme_bw() + 
  # 회색 팔레트 적용 (짙은 회색부터 시작하여 구분감 제공)
  scale_color_grey(start = 0.1, end = 0.6) +
  # 제목 및 축 라벨
  labs(
    title = "소득, 자녀수, 나이 조건별 CATE",
    x = "소득",
    y = "조건부 평균 처치효과 (CATE)",
    color = "자녀수",      # 범례 제목 통합
    linetype = "자녀수"    # 범례 제목 통합
  ) +
  # 레이아웃 및 텍스트 설정
  theme(
    legend.position = "bottom",
    strip.text = element_text(face = "bold"),
    # 분할된 그래프의 배경을 흰색으로 변경
    strip.background = element_rect(fill = "white", color = "black")
  )
